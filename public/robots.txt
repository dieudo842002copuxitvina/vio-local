# ── PARAMETERIZED DUPLICATE CONTENT ──────────────────────────────────────────
#
# Tracking parameters: fbclid, gclid, utm_* create thousands of duplicate URLs
# with zero indexable difference. Block them to protect crawl budget.
#
# Sort/filter parameters: reorder or refilter the same content — duplicates.
#
# Pagination (?page=): NOT blocked here. Page 2+ already carries
# <meta name="robots" content="noindex, follow"> and canonical → page 1
# via the thin-page engine. Crawling pagination lets Google discover
# deeper listing content through links.

Disallow: /*?utm_medium=
Disallow: /*?utm_campaign=
Disallow: /*?utm_content=
Disallow: /*?utm_term=
Disallow: /*?fbclid=
Disallow: /*?gclid=
Disallow: /*?msclkid=
Disallow: /*?ref=
Disallow: /*?sort=
Disallow: /*?filter=
Disallow: /*?tab=
